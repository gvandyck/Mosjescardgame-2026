From src/abilities/placeEffects.js — effect function signature and id-check pattern (use effect_coerts_caravan as the analog for cardId substring checks):
```js
// effect_toennoe: called with (gameState, playerId) — NOT called; it's END_PHASE so no playerId
// Actually END_PHASE effects like effect_the_gym take only (gameState). Follow that pattern.
// GANDOE check: String(mosje.cardId).toLowerCase().includes('gandoe')
// MICHELLE/TUK check: id.includes('michelle') — mosje_michelle has tags: ["MICHELLE", "TUK"]
//   but id-based check on cardId is the established engine pattern (not tag array checks)

export function effect_toennoe(gameState) {
  const state = cloneState(gameState);
  for (const playerId of Object.keys(state.players)) {
    const player = state.players[playerId];
    // Scan all active slots for GANDOE and MICHELLE Mosjes
    // Track whether both are active simultaneously for the +10 bonus
    let gandoeSlot = null;
    let michelleSlot = null;
    for (const mosje of player.activeSlots) {
      if (!mosje || mosje.isDefeated) continue;
      const id = String(mosje.cardId).toLowerCase();
      if (id.includes('gandoe')) { mosje.mp += 20; gandoeSlot = mosje; }
      else if (id.includes('michelle')) { mosje.mp += 15; michelleSlot = mosje; }
    }
    if (gandoeSlot && michelleSlot) {
      gandoeSlot.mp += 10;
      michelleSlot.mp += 10;
    }
  }
  return state;
}
```
