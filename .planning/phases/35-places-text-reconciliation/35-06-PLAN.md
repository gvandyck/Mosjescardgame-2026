+1 quest dice bonus, and a +1 buff-duration bonus — none of which the card's text ever mentioned)

`getActiveSynergies` current partner-gate (src/engine/synergyResolver.js:17-55, full function —
read before editing):
```
export function getActiveSynergies(gameState, playerId) {
	const player = gameState.players[playerId];
	if (!player) return [];
	const activeMosjeIds = player.activeSlots.filter(slot => slot !== null && !slot.isDefeated).map(slot => slot.cardId);
	const synergies = [];
	const alreadyChecked = new Set();
	for (const mosjeId of activeMosjeIds) {
		const mosjeData = MOSJES.find(m => m.id === mosjeId);
		if (!mosjeData || !mosjeData.synergyWith) continue;
		for (const partnerId of mosjeData.synergyWith) {
			const pairKey = [mosjeId, partnerId].sort().join('|');
			if (alreadyChecked.has(pairKey)) continue;
			alreadyChecked.add(pairKey);
			if (activeMosjeIds.includes(partnerId)) {   // ← the gate to bypass
				synergies.push({ mosjeAId: mosjeId, mosjeBId: partnerId, synergyEffect: mosjeData.synergyEffect });
			}
		}
	}
	return synergies;
}
```

`getPartnerSynergyQuestBonus` current gate (src/abilities/questLogic.js:52-66):
```
const PARTNER_QUEST_SYNERGIES = [{ pair: ['mosje_martin_senor_west', 'mosje_azn_cless'], category: 'Physical', bonus: 15 }];
export function getPartnerSynergyQuestBonus(gameState, playerId, category) {
	if (!category) return 0;
	const activeIds = new Set(getActiveMosjes(gameState?.players?.[playerId]).map(m => m.cardId));
	let bonus = 0;
	for (const entry of PARTNER_QUEST_SYNERGIES) {
		if (entry.category === category && entry.pair.every(id => activeIds.has(id))) { bonus += entry.bonus; }   // ← entry.pair.every(...) is the gate to soften
	}
	return bonus;
}
```
