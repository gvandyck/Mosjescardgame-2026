Replace `ability_tuk_architect_perfect_placement` with a version that accepts an optional `orderedCardIds` parameter:

```javascript
// Tuk Architect — reorder top 3 of own deck.
// orderedCardIds: optional array of up to 3 cardId strings in the desired new order.
// If provided, the top 3 cards are reordered to match. Cards not in the array stay below.
// If not provided, peek data is stored in state._architectPeek for the UI to call back.
export function ability_tuk_architect_perfect_placement(gameState, playerId, orderedCardIds = null) {
    const state = cloneState(gameState);
    const player = state.players[playerId];
    if (!player) return state;
    const top3 = player.deck.slice(0, 3);
    if (top3.length === 0) return state;

    if (Array.isArray(orderedCardIds) && orderedCardIds.length > 0) {
        // Reorder the top 3 cards to match orderedCardIds; unrecognised cards stay at bottom
        const rest = player.deck.slice(top3.length);
        const reordered = orderedCardIds
            .map(id => top3.find(c => c.cardId === id))
            .filter(Boolean);
        // Append any top3 cards not mentioned in orderedCardIds
        const mentioned = new Set(orderedCardIds);
        const leftovers = top3.filter(c => !mentioned.has(c.cardId));
        player.deck = [...reordered, ...leftovers, ...rest];
        console.log('[ABILITY] Tuk Architect: reordered top 3 →', player.deck.slice(0, 3).map(c => c.cardId));
    } else {
        // First call — just expose peek for UI to call back with orderedCardIds
        state._architectPeek = { playerId, cards: top3.map(c => c.cardId) };
        console.log('[ABILITY] Tuk Architect: peeked top 3 of own deck:', state._architectPeek.cards);
    }
    return state;
}
```

This is a two-step flow: first call with no orderedCardIds stores the peek; UI shows the player the 3 cards and calls the ability again with the chosen order.
